var targets=new[]{TARGETS};

var shots=new System.Collections.Generic.List<string>();

foreach(var r in rs)r.enabled=r==sel;

p.ambientColor=new Color(.3f,.3f,.3f);p.Render(true,false);

}
foreach(var r in rs)r.enabled=true;
}
}finally{p.Cleanup();}
return shots;
